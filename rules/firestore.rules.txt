rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ─── HELPER FUNCTIONS ────────────────────────────────────────────
    function isAuth() {
      return request.auth != null;
    }
    function isOwner(userId) {
      return isAuth() && request.auth.uid == userId;
    }
    function isAdmin() {
      return isAuth() && request.auth.token.admin == true;
    }
    function isTier(tier) {
      return isAuth() && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.profile.tier == tier;
    }
    function hasExplicitConsent(field) {
      return request.auth.token[field] == true;
    }
    function notDeleting() {
      return request.method != 'delete';
    }

    // ─── USERS ───────────────────────────────────────────────────────
    match /users/{userId} {
      allow read: if isOwner(userId) || isAdmin();
      allow create: if isAuth() && request.auth.uid == userId;
      allow update: if isOwner(userId) || isAdmin();
      allow delete: if isAdmin(); // hanya admin bisa trigger GDPR erasure
    }

    // ─── ANIMALS ─────────────────────────────────────────────────────
    match /animals/{animalId} {
      allow read: if isAuth() &&
        (resource.data.ownerId == request.auth.uid || isAdmin());
      allow create: if isAuth() &&
        request.resource.data.ownerId == request.auth.uid;
      allow update: if isAuth() &&
        resource.data.ownerId == request.auth.uid;
      allow delete: if isAdmin();
    }

    // ─── DAILY LOGS ──────────────────────────────────────────────────
    match /dailyLogs/{logId} {
      allow read: if isAuth() &&
        (resource.data.userId == request.auth.uid || isAdmin());
      allow create: if isAuth() &&
        request.resource.data.userId == request.auth.uid;
      allow update: if isAdmin(); // logs tidak bisa diubah user
      allow delete: if false;     // logs TIDAK PERNAH dihapus
    }

    // ─── HEALTH CHECKS ───────────────────────────────────────────────
    match /healthChecks/{checkId} {
      allow read: if isAuth() &&
        (resource.data.userId == request.auth.uid || isAdmin());
      allow create: if isAuth() &&
        request.resource.data.userId == request.auth.uid;
      allow update: if isAdmin();
      allow delete: if false;
    }

    // ─── ORGANIC CREDITS ─────────────────────────────────────────────
    match /organicCredits/{creditId} {
      allow read: if isAuth() &&
        (resource.data.userId == request.auth.uid || isAdmin());
      allow create: if isAuth() &&
        request.resource.data.userId == request.auth.uid;
      allow update: if isAdmin(); // hanya admin yang bisa approve/reject
      allow delete: if false;
    }

    // ─── TRANSACTIONS ────────────────────────────────────────────────
    match /transactions/{txId} {
      allow read: if isAuth() && (
        resource.data.fromUserId == request.auth.uid ||
        resource.data.toUserId == request.auth.uid ||
        isAdmin()
      );
      allow create: if isAdmin(); // transaksi hanya dibuat server-side
      allow update: if isAdmin();
      allow delete: if false;
    }

    // ─── MARKETPLACE LISTINGS ─────────────────────────────────────────
    match /marketplaceListings/{listingId} {
      allow read: if isAuth();
      allow create: if isAuth() &&
        request.resource.data.sellerId == request.auth.uid;
      allow update: if isAuth() &&
        resource.data.sellerId == request.auth.uid;
      allow delete: if isAuth() &&
        resource.data.sellerId == request.auth.uid;
    }

    // ─── AUDIT LOG — IMMUTABLE ────────────────────────────────────────
    match /auditLog/{logId} {
      allow read: if isAdmin();
      allow create: if isAdmin(); // hanya server-side via admin SDK
      allow update: if false;     // TIDAK PERNAH diubah
      allow delete: if false;     // TIDAK PERNAH dihapus
    }

    // ─── SYSTEM STATUS ───────────────────────────────────────────────
    match /system_status/{module} {
      allow read: if isAuth();
      allow write: if isAdmin();
    }

    // ─── PROCESSED WEBHOOKS (idempotency) ────────────────────────────
    match /processed_webhooks/{orderId} {
      allow read: if isAdmin();
      allow write: if isAdmin();
      allow delete: if false;
    }

    // ─── CATCH-ALL: DENY EVERYTHING ELSE ─────────────────────────────
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
